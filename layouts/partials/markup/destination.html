{{- /* The URL a link or image in a page's markdown goes to. Takes the page ("page") and the destination as written ("destination"). Content that is also read as plain files, like documentation kept in a repository, links relative to the file it is in: to "other.md", or to "../assets/run.gif". A relative link to a markdown file is a link to that page, and one to a file of the page, or of a section above it, is a link to that file. Every other link is left as it is written. */ -}} {{- $href := .destination -}} {{- $url := urls.Parse .destination -}} {{- $relative := and $url.Path (not $url.Scheme) (not $url.Host) (not (strings.HasPrefix $url.Path "/")) -}} {{- with and $relative .page.File -}} {{- $target := path.Join "/" (replace .Dir "\\" "/") $url.Path -}} {{- $found := false -}} {{- if strings.HasSuffix $url.Path ".md" -}} {{- /* The page of a directory is the directory. */ -}} {{- if in (slice "index.md" "README.md" "_index.md") (path.Base $target) -}} {{- $target = path.Dir $target -}} {{- end -}} {{- $found = site.GetPage $target -}} {{- else -}} {{- range $owner := (slice $.page) | append $.page.Ancestors -}} {{- with $owner.File -}} {{- $dir := path.Join "/" (replace .Dir "\\" "/") -}} {{- if and (not $found) (strings.HasPrefix $target (printf "%s/" $dir)) -}} {{- $found = $owner.Resources.Get (strings.TrimPrefix (printf "%s/" $dir) $target) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $found -}} {{- $href = .RelPermalink -}} {{- with $url.Fragment -}}{{- $href = printf "%s#%s" $href . -}}{{- end -}} {{- end -}} {{- end -}} {{- return $href -}}
